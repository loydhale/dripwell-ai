def invalidation_projection(body):
    need(type(body) is bytes and 0 < len(body) <= 2048, 'Invalidation byte bound')
    def pairs(items):
        value = {}
        for key, item in items: need(key not in value, 'Duplicate invalidation key'); value[key] = item
        return value
    def constant(_value): raise AssertionError('Invalidation nonfinite JSON')
    value = json.loads(body.decode('utf-8'), object_pairs_hook=pairs, parse_constant=constant)
    fields = {'version','event','event_case','active_case','event_phase','active_phase','owner_relation','bound','current_session','same_acquisition','owner_closing','global_cleanup','event_order','owner_ordinal','primary_code'}
    need(type(value) is dict and set(value) == fields, 'Invalidation exact fields')
    need(type(value['version']) is int and value['version'] == 1 and value['event'] == 'DOCUMENT_UPDATED', 'Invalidation fixed event')
    need(value['event_case'] in INVALIDATION_CASES and value['active_case'] in INVALIDATION_CASES and value['event_phase'] in INVALIDATION_PHASES and value['active_phase'] in INVALIDATION_PHASES, 'Invalidation closed case/phase')
    need(value['owner_relation'] in ['CURRENT','PREVIOUS','NONE','UNKNOWN'] and value['primary_code'] in ['BOUND_DOCUMENT_INVALIDATED','OTHER_EXISTING_PRIMARY'], 'Invalidation relation/primary')
    for k in ['bound','current_session','same_acquisition','owner_closing','global_cleanup']: need(type(value[k]) is bool, 'Invalidation flags')
    need(value['bound'] is True, 'Unbound invalidation receipt invented')
    for k in ['event_order','owner_ordinal']: need(type(value[k]) is int and 1 <= value[k] <= 4096, 'Invalidation counters')
    return value

def owned_invalidation_descriptor(item, directory, token):
    need(type(item) is dict and set(item) == {'path','bytes','sha256'} and Path(item['path']).parent == directory and type(item['bytes']) is int and 0 < item['bytes'] <= 2048, 'Owned invalidation descriptor')
    need(Path(item['path']).resolve() == Path(item['path']) and Path(item['path']).stat().st_size == item['bytes'], 'Invalidation persisted byte bound')
    body = frozen(item['path']); need(len(body) == item['bytes'] and sha(body) == item['sha256'] and token.encode() not in body, 'Invalidation hash/capability binding')
    return invalidation_projection(body)
