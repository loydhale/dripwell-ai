def need(ok, reason):

def sha(body): return hashlib.sha256(body).hexdigest()
def utc(text): return datetime.fromisoformat(text.replace('Z', '+00:00'))
def frozen(path):
    path = Path(path)
    need(path.resolve() == path and path.stat().st_mode & 0o777 == 0o400, 'Nonfrozen canonical descriptor')
    return path.read_bytes()

def pin(item, private=False):
    path = Path(item['path']); canonical = path.resolve()
    need((path.parent == BASE and canonical == path) if private else str(canonical) == item['canonical'], 'Input authorization')
    body = frozen(path) if private else canonical.read_bytes()
    need(len(body) == item['bytes'] and sha(body) == item['sha256'], 'Input changed')

def identity(pid):
    try:
        body = Path('/proc/' + str(pid) + '/stat').read_text(); f = body[body.rfind(')') + 2:].split()
        return {'pid': pid, 'ppid': int(f[1]), 'pgid': int(f[2]), 'birth': f[19]}
    except FileNotFoundError: return None

def children(pid):
    try:
        tasks = list(Path('/proc/' + str(pid) + '/task').iterdir()); need(len(tasks) <= 256, 'Owned thread bound')
        result = set()
        for task in tasks:
            try: result.update(int(v) for v in (task / 'children').read_text().split())
            except FileNotFoundError: pass
        return sorted(result)
    except FileNotFoundError: return []

def absent(group):
    try: os.killpg(group, 0); return False
    except ProcessLookupError: return True

def exclusive(path, value):
    fd = os.open(path, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o400)
    with os.fdopen(fd, 'w') as file: json.dump(value, file, indent=2)
