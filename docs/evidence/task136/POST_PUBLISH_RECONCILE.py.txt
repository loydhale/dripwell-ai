def git(*args):
    return subprocess.check_output(['git', '--no-optional-locks', '-c', 'core.fsmonitor=false',
        '-c', 'gc.auto=0', '-c', 'maintenance.auto=false', '-c', 'fetch.writeCommitGraph=false',
        '-c', 'credential.helper=', '-c', 'credential.interactive=false',
        *args], cwd=R, env={**os.environ, 'GIT_TERMINAL_PROMPT': '0'}, timeout=45)

def oid(kind, b):
    return hashlib.sha1(kind.encode() + b' ' + str(len(b)).encode() + b'\0' + b).hexdigest()

def ls_tree(commit):
    result = {}
    for row in git('ls-tree', '-r', '--full-tree', '-z', commit).split(b'\0'):
        if row:
            hdr, path = row.split(b'\t', 1)
            mode, kind, sha = hdr.decode().split()
            result[path.decode()] = (mode, kind, sha)
    return result
