def need(ok, label):
    if not ok:
        raise AssertionError(label)

def sha(body):
    return hashlib.sha256(body).hexdigest()

def exact(path, mode=0o400, maximum=1048576):
    path = Path(path)
    need(path.resolve() == path and not path.is_symlink(), 'Canonical input')
    stat = path.stat()
    need(stat.st_mode & 0o777 == mode and stat.st_size <= maximum, 'Input mode/bound')
    return path.read_bytes()

def pin(item):
    path = Path(item['path'])
    need(str(path.resolve()) == item['canonical'], 'Pinned canonical input')
    body = exact(path.resolve(), int(item['mode'], 8), 134217728)
    need(len(body) == item['bytes'] and sha(body) == item['sha256'], 'Pinned input bytes')
    return body

def exclusive(path, value):
    need(path.parent.resolve() == path.parent and not path.is_symlink(), 'Exclusive parent')
    body = json.dumps(value, indent=2, allow_nan=False).encode()
    need(len(body) <= 131072, 'Owned result bound')
    fd = os.open(path, os.O_CREAT | os.O_EXCL | os.O_WRONLY | os.O_NOFOLLOW, 0o400)
    with os.fdopen(fd, 'wb') as file:
        file.write(body)

def identity(pid):
    body = Path('/proc/' + str(pid) + '/stat').read_text()
    fields = body[body.rfind(')') + 2:].split()
    return {'pid': pid, 'pgid': int(fields[2]), 'birth': fields[19]}

def group_absent(pgid):
    try:
        os.killpg(pgid, 0)
        return False
    except ProcessLookupError:
        return True

def bounded_child_output(child, work_end, work_utc):
    end = min(time.monotonic()+25,work_end)
    buffers = {'stdout':bytearray(),'stderr':bytearray()}
    limits = {'stdout':65536,'stderr':16384}
    with selectors.DefaultSelector() as selector:
        for name, stream in [('stdout',child.stdout),('stderr',child.stderr)]:
            os.set_blocking(stream.fileno(),False)
            selector.register(stream,selectors.EVENT_READ,name)
        while selector.get_map():
            need(time.monotonic()<end and time.time()<work_utc,'Actual child output deadline')
            for key, _ in selector.select(min(0.1,max(0,end-time.monotonic()),max(0,work_utc-time.time()))):
                name=key.data
                block=os.read(key.fd,min(4096,limits[name]-len(buffers[name])+1))
                if not block:
                    selector.unregister(key.fileobj)
                else:
                    need(len(buffers[name])+len(block)<=limits[name],'Actual pipe output cap')
                    buffers[name].extend(block)
        child.wait(timeout=min(1,max(0.001,end-time.monotonic()),max(0.001,work_utc-time.time())))
    return bytes(buffers['stdout']),bytes(buffers['stderr'])
