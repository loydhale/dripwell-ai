class DocumentAcquisition {

  handle(node) {

class InvalidationReceipt {

  finish(acquisition, owner, state) {
    this.first = acquisition.first;
    let body;
    try { body = serializeInvalidation(invalidationValue(acquisition,owner,state),this.secret); }
    catch { this.outcome = {status:'FAILED',code:'INVALIDATION_RECEIPT_REJECTED'}; return; }
    try {
      this.io.writeFileSync(this.destination,body,{flag:'wx',mode:0o400});
      const persisted = frozen(this.destination); need(persisted.equals(body),'Observer persisted equality');
      this.outcome = {status:'PERSISTED',receipt:{path:this.destination,bytes:body.length,sha256:hash(body)}};
    } catch { this.outcome = {status:'FAILED',code:'INVALIDATION_RECEIPT_WRITE_FAILED'}; }
  }
}
