class DocumentAcquisition {
  constructor(session, target, url, current, transport, terminal, intentionalClose, failure) {
    this.session = session; this.target = target; this.url = url; this.current = current; this.transport = transport;

  }
  stop(reason) { if (this.first === null) { this.first = reason; this.failure(reason); } }

  check() {
    if (this.first !== null) throw new Error(this.first);

  handle(node) {

  }
}
